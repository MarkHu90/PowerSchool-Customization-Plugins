<script>
  function waitForElement(selector, callback) {
    let interval = setInterval(() => {
      let element = document.querySelector(selector);
      if (element && parseInt(element.getAttribute('touched')) !== 1) {
        element.setAttribute('touched', 1);
        //clearInterval(interval);
        callback(element);
      }
    }, 500);
  }

  waitForElement('#PSUGAsiaANAttendanceAttendanceNoticesCalendar', (e) => {
    $j.getJSON('/admin/psugasia/attendancenotices/menucount.json', function(data) {
      if (data?.count) {
        $j('span', e).append(' (<span style="color:red;font-weight:bold">'+data?.count+'</span>)');
        $j('#PSUGAsiaANAttendanceAttendanceNoticesListing span').append(' (<span style="color:red;font-weight:bold">'+data?.count+'</span>)');
      }
    });
  });
</script>
