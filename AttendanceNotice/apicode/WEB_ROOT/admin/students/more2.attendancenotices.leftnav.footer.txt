<script>
	jQuery("#std_information").append('<li><a href="attendancenotices/calendar.html?frn=~(frn)">Attendance Notices<span id="navATN_span" style="color:red; display: inline;"></span></a></li>');

  ~[tlist_sql;
  select
  count(1) as c
  
  from u_attendancenotices_requests uanr
  inner join terms y on uanr.date_start between y.firstday and y.lastday and uanr.date_end between y.firstday and y.lastday and uanr.schoolid = y.schoolid and y.isyearrec = 1
  inner join students s on uanr.studentsdcid = s.dcid
  where
  uanr.schoolid = '~(curschoolid)'
  and y.yearid = '~(curyearid)'
  and nvl(uanr.processed,0) = 0
  and s.id = '~(curstudid)';]
  if (~(c) > 0)
    jQuery("#navATN_span").html(' (~(c))');
  [/tlist_sql]

</script>
